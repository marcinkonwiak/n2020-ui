## Usage

```tsx
import {
  Field,
  FieldDescription,
  FieldError,
  FieldLabel,
} from "@/components/ui/field";
```

```tsx
<Field>
  <FieldLabel htmlFor="name">Full name</FieldLabel>
  <Input id="name" />
  <FieldDescription>The courier sees this name.</FieldDescription>
  <FieldError>Enter your full name.</FieldError>
</Field>
```
