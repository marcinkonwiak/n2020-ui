## Usage

```tsx
import {
  InputGroup,
  InputGroupAddon,
  InputGroupInput,
} from "@/components/ui/input-group";
```

```tsx
<InputGroup>
  <InputGroupAddon>
    <SearchIcon />
  </InputGroupAddon>
  <InputGroupInput placeholder="Search invoices" />
</InputGroup>
```

The `align` prop of `InputGroupAddon` sets the edge: `inline-start`, `inline-end`, `block-start`, or `block-end`.
