## Usage

```tsx
import {
  Kbd,
  KbdGroup,
} from "@/components/ui/kbd";
```

```tsx
<KbdGroup>
  <Kbd>Ctrl</Kbd>
  <Kbd>K</Kbd>
</KbdGroup>
```
