## Usage

```tsx
import {
  Collapsible,
  CollapsibleContent,
  CollapsibleTrigger,
} from "@/components/ui/collapsible";
```

```tsx
<Collapsible>
  <CollapsibleTrigger>Show variables</CollapsibleTrigger>
  <CollapsibleContent>REDIS_URL</CollapsibleContent>
</Collapsible>
```

To use another component as the trigger, pass it to the `render` prop.

```tsx
<CollapsibleTrigger render={<Button variant="ghost" />}>
  Show variables
</CollapsibleTrigger>
```
