## Usage

```tsx
import {
  Drawer,
  DrawerContent,
  DrawerDescription,
  DrawerHeader,
  DrawerTitle,
  DrawerTrigger,
} from "@/components/ui/drawer";
```

```tsx
<Drawer>
  <DrawerTrigger render={<Button variant="outline" />}>Open</DrawerTrigger>
  <DrawerContent>
    <DrawerHeader>
      <DrawerTitle>Cancel your subscription?</DrawerTitle>
      <DrawerDescription>
        Your plan stays active until 30 June.
      </DrawerDescription>
    </DrawerHeader>
  </DrawerContent>
</Drawer>
```

Set `showSwipeHandle` on `Drawer` to show a grab handle.

## Examples
