## Usage

```tsx
import {
  Tooltip,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from "@/components/ui/tooltip";
```

```tsx
<TooltipProvider>
  <Tooltip>
    <TooltipTrigger render={<Button variant="outline" />}>Archive</TooltipTrigger>
    <TooltipContent>Move this project to the archive</TooltipContent>
  </Tooltip>
</TooltipProvider>
```

`TooltipProvider` removes the open delay for the tooltips inside it. Put one at the root of your app.
